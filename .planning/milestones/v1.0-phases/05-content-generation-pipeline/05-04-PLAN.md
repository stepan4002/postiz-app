```typescript
import useSWR from 'swr';
import { useFetch } from '@gitroom/helpers/utils/custom.fetch';

export const useCompanyMedia = (companySlug: string, page = 1, tag?: string) => {
  const fetch = useFetch();
  return useSWR<CompanyMediaResponse>(
    companySlug ? `company-media-${companySlug}-p${page}${tag ? `-t${tag}` : ''}` : null,
    async () => {
      const res = await fetch(`/companies/${companySlug}/media?page=${page}${tag ? `&tag=${tag}` : ''}`);
      return res.json();
    }
  );
};
```
