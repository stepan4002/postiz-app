       a. use-post-analytics.ts:
          - 'use client'
          - import useSWR from 'swr', import { useFetch } from '@gitroom/helpers/utils/custom.fetch'
          - Define PostMetricsResponse interface locally (avoid importing from backend):
            { id, variantId, platform, snapshotType, impressions, reach, likes, comments, shares, saves, clicks, fetchedAt }
          - export const usePostAnalytics = (postId: string) => { const fetch = useFetch(); return useSWR(postId ? `post-analytics-${postId}` : null, () => fetch(`/analytics/posts/${postId}`).then(r => r.json())); }

       b. use-dashboard.ts:
          - 'use client'
          - import useSWR from 'swr', import { useFetch } from '@gitroom/helpers/utils/custom.fetch'
          - Define DashboardResponse interface locally:
            { scheduledTodayCount, pendingReviewCount, failedPostsCount, topPosts[], scheduledPosts[], failedPosts[], pendingReview[], computedAt }
          - export const useDashboard = (companySlug: string | null) => { const fetch = useFetch(); const slug = companySlug || 'all'; return useSWR(`dashboard-${slug}`, () => fetch(`/dashboard?companySlug=${slug}`).then(r => r.json()), { refreshInterval: 60000 }); }
          - refreshInterval: 60000 (1 minute) for near-real-time dashboard updates

    2. Create per-post analytics components:

       a. metrics-snapshot-row.tsx:
          - 'use client'
          - Props: { snapshotType: string, metrics: PostMetricsResponse }
          - Display: snapshotType label (1h, 24h, 7d) then 7 metric cards in a row
          - Each metric card: label (Impressions, Reach, etc.) + value (formatted number or '--' for null)
          - Use Tailwind: flex gap-2, each card: bg-card rounded-md p-2 text-center
          - Null values render as '--' (not 0) — important distinction per CONTEXT.md

       b. post-analytics-panel.tsx:
          - 'use client'
          - Props: { postId: string }
          - Uses usePostAnalytics(postId) SWR hook
          - Groups metrics by variantId, then shows each variant's metrics
          - Per variant: platform badge (colored: instagram=bg-pink-500, facebook=bg-blue-500, linkedin=bg-sky-500, x=bg-gray-500) + caption preview (truncated to 60 chars)
          - Below each variant: 3 MetricsSnapshotRow components for 1h, 24h, 7d (if available)
          - Show "No analytics data yet" if no metrics for a variant
          - Loading state: skeleton shimmer (div with animate-pulse bg-card rounded-md)
          - Error state: "Failed to load analytics"
