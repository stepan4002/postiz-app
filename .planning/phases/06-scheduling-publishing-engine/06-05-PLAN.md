       ```typescript
         const fetch = useFetch();
         return useSWR(
           companySlug ? `calendar-${companySlug}-${from}-${to}` : null,

         );
       };
       export default useScheduledPosts;
       ```

       `use-failed-posts.ts`:
       ```typescript
       const useFailedPosts = (companySlug: string, platform?: string, page?: number) => {
         const fetch = useFetch();
         const params = new URLSearchParams();
         if (platform) params.set('platform', platform);
         if (page) params.set('page', String(page));
         return useSWR(
           companySlug ? `failed-posts-${companySlug}-${platform}-${page}` : null,
           () => fetch(`/api/companies/${companySlug}/failed-posts?${params}`)
         );
       };
       export default useFailedPosts;
       ```

       `use-schedule-actions.ts` — NOT SWR, uses useFetch for mutations:
       ```typescript
       const useScheduleActions = (companySlug: string) => {
         const fetch = useFetch();
         return {
           schedule: (postId: string, scheduledAt: string, timezone?: string) =>
             fetch(`/api/companies/${companySlug}/posts/${postId}/schedule`, {
               method: 'POST', body: JSON.stringify({ scheduledAt, timezone }),
               headers: { 'Content-Type': 'application/json' }
             }),
           autoSlot: (postId: string, timezone?: string) =>
             fetch(`/api/companies/${companySlug}/posts/${postId}/auto-slot`, {
               method: 'POST', body: JSON.stringify({ timezone }),
               headers: { 'Content-Type': 'application/json' }
             }),
           cancel: (postId: string) =>
             fetch(`/api/companies/${companySlug}/posts/${postId}/schedule`, { method: 'DELETE' }),
           retry: (postId: string) =>
             fetch(`/api/companies/${companySlug}/posts/${postId}/retry`, { method: 'POST' }),
         };
       };
       export default useScheduleActions;
       ```
