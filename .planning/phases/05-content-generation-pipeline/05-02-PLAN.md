    1. Create extensions/content-generation/src/posts/content-post.service.ts:
       - @Injectable() class ContentPostService
       - Constructor: repository (ContentPostRepository), aiRouter (AIProviderRouter), aiConfigService (AIConfigService -- imported from @social/ai-service), prisma (any -- for media lookup)
       - Private resolveMediaUrl(path: string): string -- if path starts with 'http' return as-is, else prepend MINIO_PUBLIC_URL/MINIO_BUCKET
       - async generate(companyId: string, dto: CreatePostDto): Promise<{ post: ContentPost, variants: PostVariant[] }>

       Generation pipeline inside generate():

       Step 1: Create ContentPost record with status='DRAFT'
         const postId = uuidv4() (pre-generate for postId references in AI calls)
         const post = await this.repository.createPost({ id: postId, companyId, brandId: dto.brandId, mediaId: dto.mediaId, brief: dto.brief, contentType: dto.contentType, status: 'DRAFT' })

       Step 2: Image analysis (if mediaId provided)
         Look up media record: const media = await (this.prisma as any).companyMedia.findUnique({ where: { id: dto.mediaId } })
         If media exists and has a path, resolve URL and call:
           const imageAnalysis = await this.aiRouter.executeImageAnalysis(
             { companyId, brandId: dto.brandId, postId: post.id, taskType: 'analyzeImage' },
             resolvedUrl,
             IMAGE_ANALYSIS_PROMPT,
             ImageAnalysisResult  // Zod schema from types
           )
         If no media or no path, imageAnalysis = null

       Step 3: Generate base caption
         Build system prompt: buildCaptionSystemPrompt(dto.contentType, imageAnalysis?.output)
         Build user message: dto.brief || 'Generate a social media caption for this content.'
         CRITICAL: User brief goes in { role: 'user' } message, NEVER in system prompt (NF1.3)
         const baseCaptionResult = await this.aiRouter.execute(
           { companyId, brandId: dto.brandId, postId: post.id, taskType: 'generateCaption' },
           [{ role: 'system', content: systemPrompt }, { role: 'user', content: userMessage }],
           PlatformCaptionResult  // Zod schema
         )

       Step 4: Load AIConfig for gating (threshold + requireAllReview)
         const aiConfig = await this.aiConfigService.findByCompany(companyId)
         const threshold = (aiConfig as any).confidenceThreshold ?? 0.7
         const requireAllReview = (aiConfig as any).requireAllReview ?? false

       Step 5: Per-platform adaptation + scoring (parallel via Promise.allSettled)
         const platformResults = await Promise.allSettled(
           dto.platforms.map(async (platform) => {
             // Adapt caption for platform
             const adapted = await this.aiRouter.execute(
               { companyId, brandId: dto.brandId, postId: post.id, taskType: 'adaptForPlatform' },
               [{ role: 'system', content: buildPlatformAdaptationPrompt(platform, baseCaptionResult.output.caption, baseCaptionResult.output.hashtags) },
                { role: 'user', content: 'Adapt the caption for this platform.' }],
               PlatformCaptionResult
             )
             // Score the adapted caption
             const scored = await this.aiRouter.execute(
               { companyId, brandId: dto.brandId, postId: post.id, taskType: 'scoreContent' },
               [{ role: 'system', content: buildScoreContentPrompt(adapted.output.caption, platform) },
                { role: 'user', content: 'Score this caption.' }],
               ConfidenceScoreResult
             )
             // Apply confidence gating
             const variantStatus = applyConfidenceGating(scored.output.score, threshold, requireAllReview)
             return {
               platform,
               caption: adapted.output.caption,
               hashtags: adapted.output.hashtags,
               confidenceScore: scored.output.score,
               status: variantStatus,
               generatedBy: adapted.model,
             }
           })
         )

       Step 6: Collect fulfilled results, log rejected ones
         const fulfilledVariants = platformResults
           .filter(r => r.status === 'fulfilled')
           .map(r => (r as PromiseFulfilledResult<any>).value)
         // Log errors for rejected platforms but continue with partial results

       Step 7: Create PostVariant records
         const variants = await this.repository.createVariants(post.id, fulfilledVariants)

       Step 8: Update ContentPost status based on variant statuses
         const postStatus = determinePostStatus(fulfilledVariants.map(v => v.status))
         await this.repository.updatePostStatus(post.id, postStatus)

       Return { post: { ...post, status: postStatus }, variants }
